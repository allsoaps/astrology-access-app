' PURPOSE: Session Management form with mutual exclusion selection (Viewer Mode vs Event Mode)
' CALLED FROM: Main navigation or menu forms
' DEPENDENCIES: tblPeople, tblEvents, tblAssignments, tblSessions, tblImpressions
' WORKFLOW: Select Viewer OR Event (not both) - selection clears the other dropdown
' VIEWER MODE: Shows all sessions for selected viewer across all their assigned events
' EVENT MODE: Shows all assigned viewers for selected event with session status/counts


' Form-level variables to prevent circular updates when clearing selections
Private m_UpdatingViewer As Boolean
Private m_UpdatingEvent As Boolean

Private m_PersonID As Long
Private m_EventID As Long
Private m_SessionID As Long
Private m_TargetShape As String



Private Sub btnAddSession_Click()
    On Error GoTo ErrorHandler
    
'    m_PersonID = Nz(Me.lstStudents.value, 0)
    If Me.cboViewer > 0 Then
        m_PersonID = Me.lstSessions.value
        m_EventID = Me.lstSessions.Column(1)
    ElseIf Me.cboEvent > 0 Then
        m_PersonID = Me.lstSessions.Column(1)
        m_EventID = Me.lstSessions.value
    Else
        MsgBox "Please select a session to view impressions.", vbInformation, "No Selection"
        Exit Sub
    End If

    ' Check for a valid PersonID. This form is for editing, so it must exist.
    If IsNull(m_PersonID) Or m_PersonID = 0 Then
        MsgBox "Cannot add/edit sessions. Select a record first.", vbExclamation, "Select a Record Required"
        Exit Sub
    End If
    
    If IsNull(m_EventID) Or m_EventID = 0 Then
        MsgBox "Cannot add/edit sessions. Select a record first.", vbExclamation, "Select a Record Required"
        Exit Sub
    End If
    
    ' Pass the context to the Session Manager form.
    ' Format: "PersonID=123"
    Dim openArgs As String
    openArgs = "PersonID|" & m_PersonID & "|EventID|" & m_EventID
    
    ' Open the Session Manager form
    DoCmd.OpenForm "frmSessionManager", acNormal, , , , , openArgs
    

   
    Exit Sub

ErrorHandler:
    MsgBox "An error occurred opening the Session Manager: " & Err.Description, vbCritical, "Error"
End Sub

Private Sub Form_Load()
    On Error GoTo ErrorHandler
    
    ' Initialize flags
    m_UpdatingViewer = False
    m_UpdatingEvent = False
    
    ' Set up both combo boxes with initial data
    SetupViewerCombo
    SetupEventCombo
    
    ' Clear any selections initially
    Me.cboViewer = Null
    Me.cboEvent = Null
    
    m_PersonID = 0
    m_EventID = 0
    m_SessionID = 0
    
    ' Initialize results display
    Me.lblResultsHeader.Visible = False
    lstSessions.RowSource = ""
    
    Exit Sub
    
ErrorHandler:
    MsgBox "Error loading Session Search form: " & Err.Description, vbCritical, "Form Load Error"
End Sub
'=========================
' Populate List Box
'=========================
Private Sub PopulateListBox()
    On Error GoTo ErrorHandler
    
    Dim sql As String
        
    With Me.lstSessions
        .RowSourceType = "Table/Query"
        .BoundColumn = 1
        .ColumnHeads = False
        
        If Not IsNull(Me.cboViewer.value) Then
            .ColumnCount = 8
            .ColumnWidths = "0cm;.80cm;8cm;0cm;2.35cm;2.35cm;2cm;2cm"
            Me.lblHdr1.Caption = "Assigned Event(s)"
            
            sql = "SELECT " & _
              "a.PersonID, e.EventID, e.EventName, " & _
              "Nz(s.SessionID, 0) AS SessionID, " & _
              "s.SessionDate, s.SessionStartTime, i.ImpressionNbr, i.SuccessAverage " & _
              "FROM ((tblAssignments a " & _
              "LEFT JOIN tblEvents e ON a.EventID = e.EventID) " & _
              "LEFT JOIN tblSessions s ON (a.PersonID = s.PersonID AND a.EventID = s.EventID AND (s.DateDeleted IS NULL OR s.DateDeleted = #1/1/1900#))) " & _
              "LEFT JOIN (SELECT DISTINCT(SessionID), Format(AVG(Success), '0.00') AS SuccessAverage, COUNT(SketchID) AS ImpressionNbr " & _
              "FROM tblImpressions WHERE (DateDeleted IS NULL OR DateDeleted = #1/1/1900#) GROUP BY SessionID) i ON s.SessionID = i.SessionID " & _
              "WHERE a.PersonID = " & Me.cboViewer.value & " " & _
              "ORDER BY s.SessionDate DESC, s.SessionStartTime"
              
              
         ElseIf Not IsNull(Me.cboEvent.value) Then
            .ColumnCount = 8
            .ColumnWidths = "0cm;.80cm;8cm;0cm;2.35cm;2.35cm;2cm;2cm"
            
            Me.lblHdr1.Caption = "Assigned Viewer(s)"
            
            sql = "SELECT " & _
              "a.EventID, p.PersonID, p.FirstName & ' " & "' & p.LastName AS FullName, " & _
              "Nz(s.SessionID, 0) AS SessionID, " & _
              "s.SessionDate, s.SessionStartTime, " & _
              "i.ImpressionNbr, i.SuccessAverage  " & _
              "FROM ((tblAssignments a " & _
              "LEFT JOIN tblPeople p ON a.PersonID = p.PersonID) " & _
              "LEFT JOIN tblSessions s ON (a.PersonID = s.PersonID AND a.EventID = s.EventID AND (s.DateDeleted IS NULL OR s.DateDeleted = #1/1/1900#))) " & _
              "LEFT JOIN (SELECT DISTINCT(SessionID), Format(AVG(Success), '0.00') AS SuccessAverage, COUNT(SketchID) AS ImpressionNbr " & _
              "FROM tblImpressions WHERE (DateDeleted IS NULL OR DateDeleted = #1/1/1900#) GROUP BY SessionID) i ON s.SessionID = i.SessionID " & _
              "WHERE a.EventID = " & Me.cboEvent.value & " " & _
              "ORDER BY p.LastName, p.FirstName"
              
       
        End If
    
        .RowSource = sql

     End With
        
    Exit Sub
    
ErrorHandler:

End Sub

Private Sub cboViewer_AfterUpdate()
    ' PURPOSE: Clear event selection when viewer is selected (mutual exclusion)
    ' CALLED FROM: User selecting a viewer in cboViewer
    
    On Error GoTo ErrorHandler
    
    ' Prevent circular updates
    If m_UpdatingViewer Then Exit Sub
    
    m_UpdatingEvent = True
    
    ' Clear the event selection to prevent confusion
    Me.cboEvent = Null
    
    m_PersonID = 0
    m_EventID = 0
    m_SessionID = 0
    
    m_PersonID = cboViewer.value
    
    ' If we have a viewer selected, we're now in "Viewer Mode"
    ' The search will show all sessions for this viewer
    PopulateListBox
    
    m_UpdatingEvent = False
    Exit Sub
    
ErrorHandler:
    m_UpdatingEvent = False
    MsgBox "Error updating viewer selection: " & Err.Description, vbExclamation, "Selection Error"
End Sub

Private Sub cboEvent_AfterUpdate()
    ' PURPOSE: Clear viewer selection when event is selected (mutual exclusion)
    ' CALLED FROM: User selecting an event in cboEvent
    
    On Error GoTo ErrorHandler
    
    ' Prevent circular updates
    If m_UpdatingEvent Then Exit Sub
    
    m_UpdatingViewer = True
    
    ' Clear the viewer selection to prevent confusion
    Me.cboViewer = Null
    
    m_PersonID = 0
    m_EventID = 0
    m_SessionID = 0
    
    m_EventID = cboEvent.value
    
    ' If we have an event selected, we're now in "Event Mode"
    ' The search will show all viewers assigned to this event
    PopulateListBox
    
    m_UpdatingViewer = False
    Exit Sub
    
ErrorHandler:
    m_UpdatingViewer = False
    MsgBox "Error updating event selection: " & Err.Description, vbExclamation, "Selection Error"
End Sub

Private Sub btnViewAssociatedImpressions_Click()
    ' PURPOSE: Open Impressions form for the selected row in results
    ' CALLED FROM: User clicking View Associated Impressions button on main form
    
    On Error GoTo ErrorHandler
    
    ' Check if session exists and has impressions
    Dim sessionExists As Boolean
    Dim hasImpressions As Boolean
    Dim hasTargetShape As Boolean
    Dim SessionID As Long
    Dim TargetShape As String
    Dim args As String
   
    If Me.cboViewer > 0 Then
        m_PersonID = Me.lstSessions.value
        m_EventID = Me.lstSessions.Column(1)
    ElseIf Me.cboEvent > 0 Then
        m_PersonID = Me.lstSessions.Column(1)
        m_EventID = Me.lstSessions.value
    Else
        MsgBox "Please select a session to view impressions.", vbInformation, "No Selection"
        Exit Sub
    End If
        
    If Nz(Me.lstSessions.Column(3), 0) = 0 Then
'        MsgBox "No session exists for this viewer and event. Please create a session first using the Viewer or Event form.", _
'               vbExclamation, "No Session Found"
'        Exit Sub
    Else
          m_SessionID = Me.lstSessions.Column(3)
    End If
        
    SessionID = m_SessionID
    TargetShape = m_TargetShape
    
    sessionExists = (SessionID > 0)
    
    If sessionExists Then
        hasImpressions = (GetImpressionCount(SessionID) > 0)
        hasTargetShape = GetTargetShape(SessionID)
    End If
    
    ' Apply business logic
    If Not sessionExists Then
        MsgBox "No session exists for this viewer and event. Please create a session first using the Session Manager.", _
               vbExclamation, "No Session Found"
        Exit Sub
    End If
    
    If Not hasImpressions Then
'        MsgBox "A session exists but has no impressions. Please add impressions first using the Session Manager.",
'               vbExclamation , "No Impressions Found"
'        Exit Sub
    End If
    
    If Not hasTargetShape Then
        MsgBox "A session exists but has no Target Shape defined. Please add a Target Shape first using the Session Manager.", _
               vbExclamation, "No Target Shape Found"
        Exit Sub
    End If
    
    args = CStr(SessionID) & "|" & Nz(TargetShape, "")
    
    ' Open Impressions form with the SessionID
    DoCmd.OpenForm "frmImpressions", , , , , , args
    
    ' Refresh the current view after Impressions form closes
    RefreshCurrentSearch
    
    Exit Sub
    
ErrorHandler:
    MsgBox "Error viewing associated impressions: " & Err.Description, vbCritical, "View Error"
End Sub

Private Sub ClearAllSelections()
    ' PURPOSE: Clear both viewer and event selections and results
    ' CALLED FROM: Could be used by a "Clear" button or reset functionality
    
    m_UpdatingViewer = True
    m_UpdatingEvent = True
    
    Me.cboViewer = Null
    Me.cboEvent = Null
    Me.lblResultsHeader.Visible = False
    
    m_UpdatingViewer = False
    m_UpdatingEvent = False
End Sub

Private Sub RefreshCurrentSearch()
    ' PURPOSE: Refresh the current search results after data changes
    ' CALLED FROM: After Session Manager or Impressions forms close
    
    If Not IsNull(Me.cboViewer) Then
        SearchByViewer Me.cboViewer
    ElseIf Not IsNull(Me.cboEvent) Then
        SearchByEvent Me.cboEvent
    End If
    
    Me.lstSessions.Requery
    
End Sub

Private Function GetSessionID(PersonID As Long, eventID As Long) As Long
    ' PURPOSE: Get SessionID for the specified viewer and event
    ' CALLED FROM: btnViewAssociatedImpressions_Click
    ' PARAMETERS: PersonID and EventID
    ' RETURNS: SessionID if exists, 0 if not found
    
    On Error GoTo ErrorHandler
    
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim sql As String
    
    Set db = CurrentDb()
    
Debug.Print "PersonID: " & PersonID & " EventID: " & eventID
    
    If PersonID > 0 Then
        sql = "SELECT SessionID FROM tblSessions WHERE PersonID = " & PersonID
    ElseIf eventID > 0 Then
        sql = "SELECT SessionID FROM tblSessions WHERE EventID = " & eventID
    Else
        Debug.Print "Missing Session"
    End If
    
    
    Set rs = db.OpenRecordset(sql, dbOpenSnapshot)
    
    If Not rs.EOF Then
        GetSessionID = rs!SessionID
    Else
        GetSessionID = 0
    End If
    
    rs.Close
    Set rs = Nothing
    Set db = Nothing
    Exit Function
    
ErrorHandler:
    GetSessionID = 0
    If Not rs Is Nothing Then
        rs.Close
        Set rs = Nothing
    End If
    Set db = Nothing
End Function

Private Function GetImpressionCount(SessionID As Long) As Long
    ' PURPOSE: Get count of impressions for the specified session
    ' CALLED FROM: btnViewAssociatedImpressions_Click
    ' PARAMETERS: SessionID
    ' RETURNS: Count of active impressions
    
    On Error GoTo ErrorHandler
    
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim sql As String
    
    Set db = CurrentDb()
    sql = "SELECT COUNT(*) AS ImpCount FROM tblImpressions " & _
          "WHERE SessionID = " & SessionID & " " & _
          "AND (DateDeleted IS NULL OR DateDeleted = #1/1/1900#)"
    Set rs = db.OpenRecordset(sql, dbOpenSnapshot)
    
    If Not rs.EOF Then
        GetImpressionCount = rs!ImpCount
    Else
        GetImpressionCount = 0
    End If
    
    rs.Close
    Set rs = Nothing
    Set db = Nothing
    Exit Function
    
ErrorHandler:
    GetImpressionCount = 0
    If Not rs Is Nothing Then
        rs.Close
        Set rs = Nothing
    End If
    Set db = Nothing
End Function

Private Function GetTargetShape(SessionID As Long) As Boolean
    ' PURPOSE: Check that a Target Shape is saved on the Session
    ' CALLED FROM: btnViewSessions_Click
    ' PARAMETERS: SessionID
    
    On Error GoTo ErrorHandler
    
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim sql As String
    
    Set db = CurrentDb()
    sql = "SELECT TargetShape FROM tblSessions " & _
          "WHERE SessionID = " & SessionID & " " & _
          "AND (DateDeleted IS NULL OR DateDeleted = #1/1/1900#)"
    
    Set rs = db.OpenRecordset(sql, dbOpenSnapshot)
    
    If Not rs.EOF Then
        GetTargetShape = True
        m_TargetShape = Nz(rs!TargetShape, "")
    Else
        GetTargetShape = False
        m_TargetShape = ""
    End If
    
    rs.Close
    Set rs = Nothing
    Set db = Nothing
    Exit Function
    
ErrorHandler:
    GetTargetShape = False
    If Not rs Is Nothing Then
        rs.Close
        Set rs = Nothing
    End If
    Set db = Nothing
End Function

Private Sub SetupViewerCombo()
    ' PURPOSE: Initialize viewer combo with all viewers who have event assignments
    ' CALLED FROM: Form_Load and when event selection is cleared
    
    On Error GoTo ErrorHandler
    
    Dim sql As String
    
    sql = "SELECT DISTINCT p.PersonID, p.LastName, p.FirstName, " & _
          "p.LastName & ', ' & p.FirstName AS ViewerName " & _
          "FROM tblPeople p " & _
          "INNER JOIN tblAssignments a ON p.PersonID = a.PersonID " & _
          "ORDER BY p.LastName, p.FirstName"
    
    With Me.cboViewer
        .RowSource = sql
        .ColumnCount = 4
        .BoundColumn = 1        ' PersonID
        .ColumnWidths = "0in;0in;0in;2.5in"  ' Hide PersonID, LastName, FirstName; show ViewerName
        .LimitToList = True
    End With
    
    Exit Sub
    
ErrorHandler:
    MsgBox "Error setting up viewer dropdown: " & Err.Description, vbExclamation, "Setup Error"
End Sub

Private Sub SetupEventCombo()
    ' PURPOSE: Initialize event combo with all events that have viewer assignments
    ' CALLED FROM: Form_Load and when viewer selection is cleared
    
    On Error GoTo ErrorHandler
    
    Dim sql As String
    
    sql = "SELECT DISTINCT e.EventID, e.EventDate, e.EventName, " & _
          "IIf(IsNull(e.EventName) OR e.EventName='', '<no name>', e.EventName) AS txtEventName, " & _
          "Format(e.EventDate,'mm/dd/yyyy') & ' - ' & " & _
          "IIf(IsNull(e.EventName) OR e.EventName='', '<no name>', e.EventName) AS EventDisplay " & _
          "FROM tblEvents e " & _
          "INNER JOIN tblAssignments a ON e.EventID = a.EventID " & _
          "ORDER BY e.EventDate DESC, EventName"
    
    With Me.cboEvent
        .RowSource = sql
        .ColumnCount = 5
        .BoundColumn = 1        ' EventID
        .ColumnWidths = "0in;0in;0in;0in;3.5in"  ' Hide EventID, EventDate, EventName; show EventDisplay
        .LimitToList = True
    End With
    
    Exit Sub
    
ErrorHandler:
    MsgBox "Error setting up event dropdown: " & Err.Description, vbExclamation, "Setup Error"
End Sub

Private Sub SearchByViewer(ViewerPersonID As Long)
    On Error GoTo ErrorHandler
    
    Dim sql As String
    sql = "SELECT e.EventID, e.EventName, e.EventDate " & _
          "FROM tblEvents e " & _
          "INNER JOIN tblAssignments a ON e.EventID = a.EventID " & _
          "WHERE a.PersonID = " & ViewerPersonID & " " & _
          "ORDER BY e.EventDate DESC"
            
    Exit Sub
    
ErrorHandler:
    MsgBox "Error: " & Err.Description, vbCritical
End Sub

Private Sub SearchByEvent(ViewerEventID As Long)
    On Error GoTo ErrorHandler
    
    Dim sql As String
    sql = "SELECT e.EventID, e.EventName, e.EventDate " & _
          "FROM tblEvents e " & _
          "INNER JOIN tblAssignments a ON e.EventID = a.EventID " & _
          "WHERE a.EventID = " & ViewerEventID & " " & _
          "ORDER BY e.EventDate DESC"
          
    Exit Sub
    
ErrorHandler:
    MsgBox "Error: " & Err.Description, vbCritical
End Sub

Private Sub btnViewTransitChart_Click()

    ElseIf Me.cboEvent > 0 Then
        m_PersonID = Me.lstSessions.Column(1)
        m_EventID = Me.lstSessions.value
    Else
        MsgBox "Please select a session to view transit charts.", vbInformation, "No Selection"
        Exit Sub
    End If

        MsgBox "No session exists for this viewer and event. Please create a session first using the Viewer or Event form.", _
               vbExclamation, "No Session Found"
        Exit Sub
    Else
          m_SessionID = Me.lstSessions.Column(3)
    End If

    If m_SessionID = 0 Or m_PersonID = 0 Or m_EventID = 0 Then
        MsgBox "Please ensure a Viewer and Event are selected.", vbExclamation
        Exit Sub
    End If

    ' Ensure the transit data is generated
    Dim Success As Boolean
    Success = GenerateTransitChartEx("Viewer_Session", m_PersonID, m_EventID, m_SessionID)

    If Not Success Then
        MsgBox "Transit chart generation failed. Generate missing Natal/Event/Session charts automatically.", vbCritical
        Exit Sub
    End If

gPersonID = m_PersonID
gEventID = m_EventID
gSessionID = m_SessionID
gTransitType = "Viewer_Session"

    ' Pass the needed context to frmAspectGrid using openArgs
    Dim args As String
    args = m_PersonID & "|" & m_EventID & "|" & m_SessionID & "|Viewer_Session"
    DoCmd.OpenForm "frmAspectGridTabs", acNormal, , , , , args
    Exit Sub

ErrorHandler:
    MsgBox "Error in View Transit Chart: " & Err.Description, vbCritical
End Sub
