Private Sub btnViewerSession_Click()
    ' Set global variables for Viewer->Event chart context
    gPersonID = gViewerPersonID          ' The current viewer
    gEventID = gEventID                  ' Already set globally from SessionManager
    gSessionID = gSessionID              ' Already set globally from SessionManager
    gTransitType = "Viewer_Session"        ' Must match your modTransitChart constant

    ' Optionally: Generate the chart here if you want to always have the latest
    ' If chart generation is slow and charts rarely change, you can skip this or check if it exists first
    Dim chartGenerated As Boolean
    chartGenerated = GenerateTransitChartEx(gTransitType, gPersonID, gEventID, gSessionID)
    If Not chartGenerated Then
        MsgBox "Could not generate Viewer->Session transit chart. Check that all necessary charts exist.", vbExclamation
        Exit Sub
    End If
    
End Sub
